% ---------- Raccourcis (affichent les intermédiaires) ----------
\newcommand{\AddPose}[2]{%
	\ifcorrige
	\opadd[displayintermediary=all]{#1}{#2}%
	\else
	\opadd[displayintermediary=all,resultstyle=\opblank]{#1}{#2}%
	\fi
}

\newcommand{\SubPose}[2]{%
	\ifcorrige
	\opsub[displayintermediary=all]{#1}{#2}%
	\else
	\opsub[displayintermediary=all,resultstyle=\opblank,intermediarystyle=\opblank]{#1}{#2}%
	\fi
}

\newcommand{\MulPose}[2]{%
	\ifcorrige
	\opmul[displayintermediary=all]{#1}{#2}%
	\else
	\opmul[displayintermediary=all,resultstyle=\opblank,intermediarystyle=\opblank]{#1}{#2}%
	\fi
}

% Division décimale (le quotient est le "result")
\newcommand{\DivDec}[2]{%
	\ifcorrige
	\opdiv[displayintermediary=all]{#1}{#2}%
	\else
	\opdiv[displayintermediary=all,resultstyle=\opblank,intermediarystyle=\opblank]{#1}{#2}%
	\fi
}

% Division euclidienne : quotient = result ; reste = remainder
\newcommand{\DivEuc}[2]{%
	\ifcorrige
	\opidiv[displayintermediary=all]{#1}{#2}%
	\else
	\opidiv[displayintermediary=all,resultstyle=\opblank,remainderstyle=\opblank,intermediarystyle=\opblank]{#1}{#2}%
	\fi
}

% ---------- Opérandes seuls (euclidienne) ----------
% 1) Quotient & reste en trous
\newcommand{\DivEucOperandesTrousQR}[2]{%
	\opidiv[displayintermediary=none,resultstyle=\opblank,remainderstyle=\opblank]{#1}{#2}%
}
% 2) Quotient & reste invisibles (blanc)
\newcommand{\DivEucOperandesBlancQR}[2]{%
	\opidiv[displayintermediary=none,resultstyle=\color{white},remainderstyle=\color{white}]{#1}{#2}%
}
% 3) Seulement le quotient en trous
\newcommand{\DivEucOperandesTrousQ}[2]{%
	\opidiv[displayintermediary=none,resultstyle=\opblank]{#1}{#2}%
}
% 4) Seulement le reste en trous
\newcommand{\DivEucOperandesTrousR}[2]{%
	\opidiv[displayintermediary=none,remainderstyle=\opblank]{#1}{#2}%
}

% ---------- Opérandes seuls (décimale) ----------
\newcommand{\DivDecOperandesTrous}[2]{%
	\opdiv[displayintermediary=none,resultstyle=\opblank]{#1}{#2}%
}
\newcommand{\DivDecOperandesBlanc}[2]{%
	\opdiv[displayintermediary=none,resultstyle=\color{white}]{#1}{#2}%
}

    \ifcorrige
        {\color{mainBlue}#2}%
    \else
        \makebox[#1]{\rule{0pt}{1.2ex}\dotfill}%
    \fi
}

% Variante auto-ajustée à la largeur naturelle du contenu (avec marge optionnelle)
% Syntaxe : \solutionauto{réponse} ou \solutionauto[5mm]{réponse}
\newsavebox{\solbox}
\newcommand{\solutionauto}[2][0pt]{%
    \ifcorrige
        {\color{mainBlue}#2}%
    \else
        \sbox{\solbox}{#2}%
        \makebox[\dimexpr\wd\solbox + #1\relax]{\rule{0pt}{1.2ex}\dotfill}%
    \fi
}
