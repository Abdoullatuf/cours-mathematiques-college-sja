	\item La demi-droite d'origine $P$ passant par $Q$ : \ans{$[PQ)$}

	\item La longueur du segment d'extrémités $T$ et $U$ : \ans{$TU$ ou $UT$}

	\item Les points $X$, $Y$ et $Z$ sont alignés : \ans{$X \in (YZ)$ ou $Y \in (XZ)$ ou $Z \in (XY)$}

				\draw[thick] (-2,0) -- (4,0);

				\draw[thick] (1,-1) -- (1,3);

%				\node[blue, thick] at (2.5,0.1) {$>$};
			\end{tikzpicture}
			% === FIN DU CODE ===
		\end{center}
	\end{minipage}
	
	\exspace
	
	\textbf{1.} Que peut-on dire des droites $(d_2)$ et $(\Delta)$ ? Justifie avec une propriété. \points{1,5}
	
	\anslines{4}{
		On sait que $(d_1) \parallel (d_2)$ et que $(\Delta) \perp (d_1)$. \\
		\textbf{Or, si deux droites sont parallèles, toute droite perpendiculaire à l'une est perpendiculaire à l'autre.} \\
		Donc, les droites $(d_2)$ et $(\Delta)$ sont \textbf{perpendiculaires}.
	}
	
	\exspace
	
	\textbf{2.} On ajoute une droite $(d_3)$ qui est perpendiculaire à la droite $(\Delta)$. \\
	Que peux-tu en déduire pour les droites $(d_1)$ et $(d_3)$ ? Justifie avec une propriété. \points{1,5}
	
	\anslines{4}{
		On sait que $(d_1) \perp (\Delta)$ et que $(d_3) \perp (\Delta)$. \\
		\textbf{Or, si deux droites sont perpendiculaires à une même droite, alors elles sont parallèles entre elles.} \\
		Donc, les droites $(d_1)$ et $(d_3)$ sont \textbf{parallèles}.
	}
	
\end{exercisebox}
